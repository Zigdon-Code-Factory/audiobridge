cmake_minimum_required(VERSION 3.20)
project(AudioBridgeTests LANGUAGES CXX)
enable_testing()
add_executable(playback_frame_test playback_frame_test.cpp)
target_include_directories(playback_frame_test PRIVATE ../common)
target_compile_features(playback_frame_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(playback_frame_test PRIVATE /UNDEBUG)
else()
    target_compile_options(playback_frame_test PRIVATE -UNDEBUG)
endif()
add_test(NAME playback_frame COMMAND playback_frame_test)

if(WIN32)
    # Manual-only hardware check; not registered with CTest because it requires
    # a selected recording endpoint and a live phone/source.
    add_executable(audio_endpoint_probe audio_endpoint_probe.cpp ../windows/src/audio_capture.cpp)
    target_include_directories(audio_endpoint_probe PRIVATE ../common)
    target_compile_features(audio_endpoint_probe PRIVATE cxx_std_17)
    target_compile_definitions(audio_endpoint_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(audio_endpoint_probe PRIVATE ole32 uuid avrt)
    add_executable(microphone_test microphone_test.cpp ../windows/src/virtual_microphone.cpp)
    target_compile_features(microphone_test PRIVATE cxx_std_17)
    target_compile_definitions(microphone_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(microphone_test PRIVATE /UNDEBUG)
    target_link_libraries(microphone_test PRIVATE advapi32)
    add_test(NAME microphone COMMAND microphone_test)
    set_tests_properties(microphone PROPERTIES TIMEOUT 20)
endif()

# Use the same mbedTLS source as the apps; no separate crypto implementation.
set(MBEDTLS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../windows/build/_deps/mbedtls-src" CACHE PATH "mbedTLS source checkout")
if(EXISTS "${MBEDTLS_SOURCE_DIR}/CMakeLists.txt")
    enable_language(C)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${MBEDTLS_SOURCE_DIR}" mbedtls)
    add_executable(secure_transport_test secure_transport_test.cpp ../common/dtls_session.cpp)
    target_include_directories(secure_transport_test PRIVATE ../common)
    target_compile_features(secure_transport_test PRIVATE cxx_std_17)
    target_link_libraries(secure_transport_test PRIVATE mbedtls mbedcrypto mbedx509)
    if(WIN32)
        target_compile_definitions(secure_transport_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WINSOCK_DEPRECATED_NO_WARNINGS)
        target_link_libraries(secure_transport_test PRIVATE ws2_32)
    endif()
    if(MSVC)
        target_compile_options(secure_transport_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(secure_transport_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME secure_transport COMMAND secure_transport_test)
    set_tests_properties(secure_transport PROPERTIES TIMEOUT 30)
    if(WIN32)
        add_executable(server_security_test server_security_test.cpp ../windows/src/network.cpp ../common/dtls_session.cpp)
        target_include_directories(server_security_test PRIVATE ../common ../windows/src)
        target_compile_features(server_security_test PRIVATE cxx_std_17)
        target_compile_definitions(server_security_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WINSOCK_DEPRECATED_NO_WARNINGS)
        target_compile_options(server_security_test PRIVATE /UNDEBUG)
        target_link_libraries(server_security_test PRIVATE mbedtls mbedcrypto mbedx509 ws2_32 iphlpapi qwave)
        add_test(NAME server_security COMMAND server_security_test)
        set_tests_properties(server_security PROPERTIES TIMEOUT 30)
    endif()
endif()
