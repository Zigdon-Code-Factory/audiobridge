cmake_minimum_required(VERSION 3.20)
project(AudioBridgeTests LANGUAGES CXX)
enable_testing()
add_executable(playback_frame_test playback_frame_test.cpp)
target_include_directories(playback_frame_test PRIVATE ../common)
target_compile_features(playback_frame_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(playback_frame_test PRIVATE /UNDEBUG)
else()
    target_compile_options(playback_frame_test PRIVATE -UNDEBUG)
endif()
add_test(NAME playback_frame COMMAND playback_frame_test)
